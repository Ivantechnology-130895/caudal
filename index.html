CREDENCIALES DE RED Y FIREBASE // ========================================== #define WIFI_SSID "FLIA CANAVIRI" #define WIFI_PASSWORD "3521422OR" #define DATABASE_URL "https://base-de-datos-ivan-default-rtdb.firebaseio.com/" #define DATABASE_SECRET "AIzaSyC408VpxKuX-TCluBuOmSEh0IwEhLUBI5o" // Configuración horaria NTP (UTC-4 para Bolivia = -14400 segundos) #define GMT_OFFSET_SEC -14400 #define DAYLIGHT_OFFSET_SEC 0 // ========================================== // 2. HARDWARE, CALIBRACIÓN Y DETECCIÓN DE AIRE // ========================================== const uint8_t PIN_SENSOR = 18; // Ajusta este factor según tu prueba volumétrica: // Si pasaste 1.00 L real y marcó 1.79 L con 7.5, el factor efectivo es ~ 7.5 * 1.79 = 13.4 // (Primero verifica si hay ruido extra; si no hay ruido extra, prueba ~13.4 o calibra con 2-5 L) float FACTOR_CALIBRACION = 13.4; // Frecuencia mínima (Hz) para ignorar 1 pulso aislado por ruido/goteo no medible const float FRECUENCIA_MIN_AGUA_HZ = 1.0; // Detección de bolsa de aire: // - Umbral absoluto alto (Hz) // - Salto brusco relativo respecto al promedio reciente (ej. 1.85x) const float FRECUENCIA_ALTA_AIRE_HZ = 110.0; const float MULTIPLICADOR_SALTO_AIRE = 1.85; const uint8_t MUESTRAS_AIRE_REQUERIDAS = 2; // Antirrebote en microsegundos (6000 us = 6 ms -> filtra rebotes/ruido rápido) const uint32_t TIEMPO_MIN_ENTRE_PULSOS_US = 6000; // Si es true, no suma al volumen acumulado mientras haya alerta de aire confirmada const bool NO_ACUMULAR_EN_AIRE = true; volatile uint32_t contadorPulsos = 0; volatile uint32_t ultimoPulsoUs = 0; uint32_t tiempoAnteriorCalculo = 0; uint32_t tiempoAnteriorEnvio = 0; uint8_t contadorMuestrasAire = 0; float freqPromedio = 0.0; float caudalLPM = 0.0; double volumenTotalLitros = 0.0; double volumenTotalM3 = 0.0; bool alertaAire = false; // ========================================== // 3. OBJETOS DE FIREBASE // ========================================== FirebaseData fbdo; FirebaseAuth auth; FirebaseConfig config; // ========================================== // 4. INTERRUPCIÓN POR HARDWARE EN IRAM // ========================================== void IRAM_ATTR contarPulsoISR() { uint32_t ahoraUs = micros(); if ((ahoraUs - ultimoPulsoUs) >= TIEMPO_MIN_ENTRE_PULSOS_US) { contadorPulsos++; ultimoPulsoUs = ahoraUs; } } // ========================================== // 5. CONFIGURACIÓN INICIAL (SETUP) // ========================================== void setup() { Serial.begin(115200); delay(1000); pinMode(PIN_SENSOR, INPUT); attachInterrupt(digitalPinToInterrupt(PIN_SENSOR), contarPulsoISR, RISING); Serial.print("Conectando a Wi-Fi: "); Serial.println(WIFI_SSID); WiFi.begin(WIFI_SSID, WIFI_PASSWORD); while (WiFi.status() != WL_CONNECTED) { delay(500); Serial.print("."); } Serial.println(""); Serial.print("WiFi Conectado! IP: "); Serial.println(WiFi.localIP()); Serial.print("Sincronizando hora con servidor NTP..."); configTime(GMT_OFFSET_SEC, DAYLIGHT_OFFSET_SEC, "pool.ntp.org", "time.nist.gov"); struct tm timeinfo; while (!getLocalTime(&timeinfo)) { Serial.print("."); delay(1000); } Serial.println("\nHora sincronizada exitosamente."); config.database_url = DATABASE_URL; config.signer.tokens.legacy_token = DATABASE_SECRET; config.timeout.serverResponse = 10 * 1000; config.timeout.socketConnection = 10 * 1000; Firebase.begin(&config, &auth); Firebase.reconnectWiFi(true); } // ========================================== // 6. BUCLE PRINCIPAL (LOOP) // ========================================== void loop() { uint32_t tiempoActual = millis(); // TAREA A: Cálculo de caudal, volumen y detección de aire (cada 1 segundo) if (tiempoActual - tiempoAnteriorCalculo >= 1000) { noInterrupts(); uint32_t pulsosEnSegundo = contadorPulsos; contadorPulsos = 0; interrupts(); float frecuenciaHz = (float)pulsosEnSegundo; // Evaluar si hay indicio de bolsa de aire (frecuencia alta o salto brusco sobre el promedio) bool saltoBrusco = (freqPromedio > 3.0) && (frecuenciaHz > (freqPromedio * MULTIPLICADOR_SALTO_AIRE)) && (frecuenciaHz > 25.0); bool superaLimiteAlto = (frecuenciaHz > FRECUENCIA_ALTA_AIRE_HZ); if (saltoBrusco || superaLimiteAlto) { contadorMuestrasAire++; if (contadorMuestrasAire >= MUESTRAS_AIRE_REQUERIDAS) { alertaAire = true; } } else { contadorMuestrasAire = 0; alertaAire = false; } // Actualizar promedio exponencial suave de frecuencia if (frecuenciaHz < FRECUENCIA_MIN_AGUA_HZ) { freqPromedio = freqPromedio * 0.5; if (freqPromedio < 0.2) freqPromedio = 0.0; caudalLPM = 0.0; } else { if (freqPromedio == 0.0) { freqPromedio = frecuenciaHz; } else { freqPromedio = (freqPromedio * 0.7) + (frecuenciaHz * 0.3); } caudalLPM = frecuenciaHz / FACTOR_CALIBRACION; // Acumular volumen solo si no está bloqueado por alerta de aire confirmada if (!(NO_ACUMULAR_EN_AIRE && alertaAire)) { double litrosEnSegundo = caudalLPM / 60.0; volumenTotalLitros += litrosEnSegundo; volumenTotalM3 = volumenTotalLitros / 1000.0; } else { caudalLPM = 0.0; } } Serial.printf("Freq: %.1f Hz (Prom: %.1f) | Caudal: %.2f L/min | Acum: %.3f L | Aire: %s\n", frecuenciaHz, freqPromedio, caudalLPM, volumenTotalLitros, alertaAire ? "SI" : "NO"); tiempoAnteriorCalculo = tiempoActual; } // TAREA B: Envío a Firebase (cada 5 segundos) if (tiempoActual - tiempoAnteriorEnvio >= 5000) { tiempoAnteriorEnvio = tiempoActual; if (WiFi.status() == WL_CONNECTED && Firebase.ready()) { FirebaseJson serverTimestamp; serverTimestamp.set(".sv", "timestamp"); FirebaseJson json; json.set("caudal_lpm", caudalLPM); json.set("volumen_litros", volumenTotalLitros); json.set("volumen_m3", volumenTotalM3); json.set("aire_detectado", alertaAire); json.set("ultima_actualizacion", serverTimestamp); if (Firebase.RTDB.setJSON(&fbdo, "/medidor_01/tiempo_real", &json)) { Serial.println("[Firebase] /medidor_01/tiempo_real actualizado"); } else { Serial.print("[Firebase Error tiempo_real] "); Serial.println(fbdo.errorReason().c_str()); } if (caudalLPM > 0.0 || alertaAire) { if (Firebase.RTDB.pushJSON(&fbdo, "/medidor_01/historial", &json)) { Serial.println("[Firebase] Registro agregado a /medidor_01/historial"); } else { Serial.print("[Firebase Error historial] "); Serial.println(fbdo.errorReason().c_str()); } } } else { Serial.println("[Estado] Esperando conexion Wi-Fi o sesion Firebase lista..."); } } }
